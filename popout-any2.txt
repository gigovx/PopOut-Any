def resource_path(rel_path: str) -> str:
    """
    Get absolute path to a resource. Works for both dev and when bundled (via PyInstaller).
    """
    base = getattr(sys, "_MEIPASS", os.path.dirname(__file__))
    return os.path.join(base, rel_path)

def enum_visible_windows():
    """
    Enumerate visible windows with a title and a system menu.
    Skips child windows and some OS helper windows.
    """
    wins = []
    def callback(hwnd, _):
        if not win32gui.IsWindowVisible(hwnd):
            return
        if win32gui.GetParent(hwnd) != 0:
            return
        cls = win32gui.GetClassName(hwnd)
        if cls in ("Progman", "Shell_TrayWnd", "Windows.UI.Core.CoreWindow"):
            return
        style = win32gui.GetWindowLong(hwnd, win32con.GWL_STYLE)
        if not (style & win32con.WS_CAPTION and style & win32con.WS_SYSMENU):
            return
        l, t, r, b = win32gui.GetWindowRect(hwnd)
        if (r - l) < 100 or (b - t) < 40:
            return
        title = win32gui.GetWindowText(hwnd)
        if title:
            wins.append((hwnd, title))
    win32gui.EnumWindows(callback, None)
    return wins

class MonitorEdgeWidget(QWidget):
    # Emits (screen, monitor_number, segment) upon an edge selection.
    edgeSelected = pyqtSignal(object, int, str)

    def __init__(self, screen, monitor_number, thumbnail_size: QSize = QSize(300, 200), parent=None):
        super().__init__(parent)
        self.screen = screen
        self.monitor_number = monitor_number
        self.thumbnail_size = thumbnail_size  # Fixed size for all monitors
        self.selected_edge = None

        self.setFixedSize(thumbnail_size)
        self.label = QLabel(self)
        # Load a base image – ensure that "monitor_base.png" is available.
        pix = QPixmap(resource_path("monitor_base.png"))
        scaled_pix = pix.scaled(thumbnail_size, Qt.AspectRatioMode.KeepAspectRatio,
                                  Qt.TransformationMode.SmoothTransformation)
        self.label.setPixmap(scaled_pix)
        self.label.setFixedSize(thumbnail_size)
        self.label.setGeometry(0, 0, thumbnail_size.width(), thumbnail_size.height())

        # Overlay a number label for identification.
        self.number_label = QLabel(f"Monitor {self.monitor_number}", self)
        self.number_label.setStyleSheet("background-color: rgba(255,255,255,0.6); font-weight: bold;")
        self.number_label.move(10, 10)

        self.segment_buttons = {}
        self._create_segment_buttons()

    def _create_segment_buttons(self):
        w = self.thumbnail_size.width()
        h = self.thumbnail_size.height()
        thickness = int(h * 0.10)
        half_h = h // 2

        def add_button(name, x, y, width_, height_):
            btn = QPushButton(self.label)
            btn.setGeometry(x, y, width_, height_)
            btn.setObjectName(name)
            btn.setStyleSheet("""
                QPushButton { background: rgba(0,0,0,0); border: none; }
                QPushButton:hover { background: rgba(0,174,239,0.2); }
            """)
            btn.clicked.connect(lambda _, s=name: self.on_segment_clicked(s))
            self.segment_buttons[name] = btn

        add_button("Top-Left", 0, 0, w // 3, thickness)
        add_button("Top-Right", 2 * (w // 3), 0, w - 2 * (w // 3), thickness)
        add_button("Bottom-Left", 0, h - thickness, w // 3, thickness)
        add_button("Bottom-Right", 2 * (w // 3), h - thickness, w - 2 * (w // 3), thickness)
        add_button("Left-Top", 0, thickness, thickness, half_h - thickness)
        add_button("Left-Bottom", 0, half_h, thickness, half_h)
        add_button("Right-Top", w - thickness, thickness, thickness, half_h - thickness)
        add_button("Right-Bottom", w - thickness, half_h, thickness, half_h)

    def on_segment_clicked(self, segment_name):
        self.selected_edge = segment_name
        # Highlight the selected button.
        for name, btn in self.segment_buttons.items():
            if name == segment_name:
                btn.setStyleSheet("background-color: rgba(0,174,239,0.3); border: none;")
            else:
                btn.setStyleSheet("""
                    QPushButton { background: rgba(0,0,0,0); border: none; }
                    QPushButton:hover { background: rgba(0,174,239,0.2); }
                """)
        self.edgeSelected.emit(self.screen, self.monitor_number, segment_name)
        print(f"Monitor {self.monitor_number} selected edge: {segment_name}")


class MonitorEdgeSelectionWidget(QWidget):

    def __init__(self, parent=None):
        super().__init__(parent)
        self.selected_monitor = None
        self.selected_edge = None
        self.monitorEdgeWidgets = []

        screens = QGuiApplication.screens()
        self.bounding_rect = self.compute_bounding_rect(screens)
        thumbnail_size = QSize(300, 200)  # Fixed size for all monitors

        # Set maximum allowed overall dimensions.
        max_disp_width = 800
        max_disp_height = 600

        # Compute the scale factor so that:
        # (bounding_rect.width * s) + thumbnail_width <= max_disp_width
        # (bounding_rect.height * s) + thumbnail_height <= max_disp_height
        s = min((max_disp_width - thumbnail_size.width()) / self.bounding_rect.width(),
                (max_disp_height - thumbnail_size.height()) / self.bounding_rect.height())

        disp_width = int(self.bounding_rect.width() * s + thumbnail_size.width())
        disp_height = int(self.bounding_rect.height() * s + thumbnail_size.height())
        self.setFixedSize(disp_width, disp_height)

        # Position each monitor widget absolutely.
        for i, screen in enumerate(screens, start=1):
            geo = screen.geometry()
            widget = MonitorEdgeWidget(screen, i, thumbnail_size=thumbnail_size, parent=self)
            # Determine the relative position using the computed scale factor.
            x = int((geo.x() - self.bounding_rect.x()) * s)
            y = int((geo.y() - self.bounding_rect.y()) * s)
            widget.move(x, y)
            widget.show()
            widget.edgeSelected.connect(self.on_edge_selected)
            self.monitorEdgeWidgets.append(widget)

    def compute_bounding_rect(self, screens):
        rect = screens[0].geometry()
        for scr in screens[1:]:
            rect = rect.united(scr.geometry())
        return rect

    def on_edge_selected(self, screen, monitor_number, segment):
        self.selected_monitor = screen
        self.selected_edge = segment
        print(f"Active target: Monitor {monitor_number}, Edge: {segment}")

class Slider:
    """Animate a window with optional easing and fade effects."""
    def __init__(self, hwnd, start, end, steps, interval,
                 use_ease=False, use_fade=False,
                 into_view=False, activate_on_show=False, cb=None):
        self.hwnd = hwnd
        self.sx, self.sy = start
        self.ex, self.ey = end
        self.steps = max(1, steps)
        self.interval = interval
        self.use_ease = use_ease
        self.use_fade = use_fade
        self.into_view = into_view
        self.activate_on_show = activate_on_show
        self.cb = cb
        self.i = 0
        if use_fade:
            ex_style = win32gui.GetWindowLong(hwnd, win32con.GWL_EXSTYLE)
            win32gui.SetWindowLong(hwnd, win32con.GWL_EXSTYLE, ex_style | win32con.WS_EX_LAYERED)
        self.timer = QTimer()
        self.timer.setInterval(self.interval)
        self.timer.timeout.connect(self._step)

    def start(self):
        self.i = 0
        self.timer.start()

    def _step(self):
        self.i += 1
        t = self.i / self.steps
        if self.use_ease:
            t = t * t * (3 - 2 * t)
        nx = int(self.sx + (self.ex - self.sx) * t)
        ny = int(self.sy + (self.ey - self.sy) * t)
        flags = win32con.SWP_NOSIZE | win32con.SWP_NOZORDER | win32con.SWP_NOACTIVATE
        win32gui.SetWindowPos(self.hwnd, None, nx, ny, 0, 0, flags)
        if self.use_fade:
            alpha = int(255 * t) if self.into_view else int(255 * (1 - t))
            win32gui.SetLayeredWindowAttributes(self.hwnd, 0, alpha, win32con.LWA_ALPHA)
        if self.i >= self.steps:
            self.timer.stop()
            if self.activate_on_show and self.into_view:
                win32gui.SetWindowPos(self.hwnd, None, self.ex, self.ey, 0, 0,
                                      win32con.SWP_NOSIZE | win32con.SWP_NOACTIVATE)
                try:
                    win32gui.SetForegroundWindow(self.hwnd)
                except Exception:
                    pass
            if self.cb:
                self.cb()

        def add_button(name, x, y, width, height):
            btn = QPushButton(self.image_label)
            btn.setGeometry(x, y, width, height)
            btn.setObjectName(name)
            btn.setStyleSheet(
                "QPushButton {background-color: rgba(0,0,0,0);} "
                "QPushButton:hover {background-color: rgba(0,174,239,0.2);}"
            )
            btn.clicked.connect(lambda _, s=name: self.on_edge_clicked(s))
            return btn
        
        self.edge_buttons = {}
        # Top edge divided into three segments.
        seg_w = w // 3
        self.edge_buttons["Top-Left"] = add_button("Top-Left", 0, 0, seg_w, thickness)
        self.edge_buttons["Top-Middle"] = add_button("Top-Middle", seg_w, 0, seg_w, thickness)
        self.edge_buttons["Top-Right"] = add_button("Top-Right", 2 * seg_w, 0, w - 2 * seg_w, thickness)
        
        # Bottom edge divided into three segments.
        self.edge_buttons["Bottom-Left"] = add_button("Bottom-Left", 0, h - thickness, seg_w, thickness)
        self.edge_buttons["Bottom-Middle"] = add_button("Bottom-Middle", seg_w, h - thickness, seg_w, thickness)
        self.edge_buttons["Bottom-Right"] = add_button("Bottom-Right", 2 * seg_w, h - thickness, w - 2 * seg_w, thickness)

        left_width = int(w * 0.10)
        left_height = (h - 2 * thickness) // 2

        self.edge_buttons["Left-Bottom"] = add_button("Left-Bottom", 0, thickness + left_height, left_width, left_height)

        right_width = left_width
        self.edge_buttons["Right-Top"] = add_button("Right-Top", w - right_width, thickness, right_width, left_height)

        for name, btn in self.edge_buttons.items():
            if name == segment:
                btn.setStyleSheet("background-color: rgba(0,174,239,0.3); border: none;")
            else:
                btn.setStyleSheet(
                    "QPushButton {background-color: rgba(0,0,0,0);} "
                    "QPushButton:hover {background-color: rgba(0,174,239,0.2);}"
                )
        current_monitor = self.current_monitor()
        monitor_index = self.monitor_combo.currentIndex() + 1
        # Emit selection.
        self.edgeSelected.emit(current_monitor, monitor_index, segment)
        print(f"Selected edge: Monitor {monitor_index} - {segment}")

        return self.monitor_combo.currentData()
    
    def current_edge(self):
        # Returns the name of the highlighted edge button (if any).
        for name, btn in self.edge_buttons.items():
            if "rgba(0,174,239,0.3)" in btn.styleSheet():
                return name
        return ""

        if reason == QSystemTrayIcon.ActivationReason.Trigger:
            self.showNormal()
            self.activateWindow()
    
    def _on_sel(self):
        has_win = bool(self.win_list.selectedItems())
        has_drop = (self.monitor_edge_selector.current_monitor() is not None and
                    self.monitor_edge_selector.current_edge() != "")
        self.assign_btn.setEnabled(has_win and has_drop)
        self.remove_btn.setEnabled(bool(self.assign_list.selectedItems()) and not self.enable_btn.isChecked())
    
    def _populate_windows(self):
        self.win_list.clear()
        # build set of already assigned HWNDs
        assigned = { cfg["hwnd"]
                     for lst in self.window_cfg.values()
                     for cfg in lst }
        for hwnd, title in enum_visible_windows():
            if hwnd in assigned:
                continue      # skip windows already assigned
            self.win_list.addItem(f"{hwnd} | {title}")
        self._on_sel()

    
    def _assign(self):
        item = self.win_list.currentItem()
        if not item:
            return
        hwnd = int(item.text().split("|", 1)[0].strip())
        # Prevent duplicates across all segments:
        if any(cfg["hwnd"] == hwnd
               for lst in self.window_cfg.values()
               for cfg in lst):
            return
        monitor_obj = self.monitor_edge_selector.current_monitor()
        monitor_num = None
        for idx, screen in enumerate(QGuiApplication.screens(), start=1):
            if screen == monitor_obj:
                monitor_num = idx
                break
        if monitor_num is None:
            return
        seg = self.monitor_edge_selector.current_edge()
        key = (monitor_num, seg)
        l, t, r, b = win32gui.GetWindowRect(hwnd)
        cfg = {
            "hwnd": hwnd,
            "orig": (l, t, r, b),
            "steps": self.steps.value(),
            "interval": self.speed.value(),
            "ease": self.chk_ease.isChecked(),
            "fade": self.chk_fade.isChecked(),
            "animating": False
        }
        self.window_cfg.setdefault(key, [])
        if not any(c["hwnd"] == hwnd for c in self.window_cfg[key]):
            self.window_cfg[key].append(cfg)
        self._refresh_assignments()
    
    def _refresh_assignments(self):
        self.assign_list.clear()
        for (mn, seg), lst in self.window_cfg.items():
            for c in lst:
                title = win32gui.GetWindowText(c["hwnd"])
                self.assign_list.addItem(f"Monitor {mn} | {seg} | {title} | {c['hwnd']}")
        self._on_sel()
        if self.window_cfg:
            self.enable_btn.setEnabled(True)
        else:
            self.enable_btn.setEnabled(False)
    
    def _on_assign_sel(self):
        self.remove_btn.setEnabled(bool(self.assign_list.selectedItems()) and not self.enable_btn.isChecked())
    
    def _remove_assignment(self):
        sel = self.assign_list.currentItem()
        if not sel:
            return
        # Expected format: "Monitor {mn} | {seg} | {title} | {hwnd}"
        parts = [p.strip() for p in sel.text().split("|")]
        monitor_text, seg_text, _, hwnd_str = parts
        mn = int(monitor_text.replace("Monitor", "").strip())
        key = (mn, seg_text)
        hwnd = int(hwnd_str)
        if key in self.window_cfg:
            self.window_cfg[key] = [c for c in self.window_cfg[key] if c["hwnd"] != hwnd]
            if not self.window_cfg[key]:
                del self.window_cfg[key]
        self._refresh_assignments()
    
    def _on_enable(self, on: bool):
        if on:
            self.triggered_segments = set()
            self.segment_trigger_time = {}
            for key, lst in self.window_cfg.items():
                for cfg in lst:
                    self._hide_taskbar(cfg["hwnd"])
                    self._slide_cfg(cfg, key[1], into_view=False)
            self.edge_timer.start()
            self.focus_timer.start()
            self.remove_btn.setEnabled(False)
        else:
            self.edge_timer.stop()
            self.focus_timer.stop()
            def _restore(cfg):
                self._show_taskbar(cfg["hwnd"])
                win32gui.SetWindowPos(
                    cfg["hwnd"], win32con.HWND_NOTOPMOST,
                    0, 0, 0, 0,
                    win32con.SWP_NOMOVE | win32con.SWP_NOSIZE | win32con.SWP_NOACTIVATE
                )
            for key, lst in self.window_cfg.items():
                for cfg in lst:
                    self._slide_cfg(cfg, key[1], into_view=True, cb=lambda cfg=cfg: _restore(cfg))
            self.triggered_segments.clear()
    
    def _hide_taskbar(self, hwnd):
        ex = win32gui.GetWindowLong(hwnd, win32con.GWL_EXSTYLE)
        ex = (ex & ~win32con.WS_EX_APPWINDOW) | win32con.WS_EX_TOOLWINDOW
        win32gui.SetWindowLong(hwnd, win32con.GWL_EXSTYLE, ex)
        win32gui.SetWindowPos(
            hwnd, None, 0, 0, 0, 0,
            win32con.SWP_NOMOVE | win32con.SWP_NOSIZE |
            win32con.SWP_NOZORDER | win32con.SWP_FRAMECHANGED
        )
    
    def _show_taskbar(self, hwnd):
        ex = win32gui.GetWindowLong(hwnd, win32con.GWL_EXSTYLE)
        ex = (ex | win32con.WS_EX_APPWINDOW) & ~win32con.WS_EX_TOOLWINDOW
        win32gui.SetWindowLong(hwnd, win32con.GWL_EXSTYLE, ex)
        win32gui.SetWindowPos(
            hwnd, None, 0, 0, 0, 0,
            win32con.SWP_NOMOVE | win32con.SWP_NOSIZE |
            win32con.SWP_NOZORDER | win32con.SWP_FRAMECHANGED
        )
    
    
    def _compute_positions(self, hwnd, dir_idx, orig):
        """
        Returns (hide_pos, orig_pos), where hide_pos is just outside the union_rect.
        """
        l, t, r, b = orig
        w, h = r - l, b - t
        pad = 2

        ux, uy, uw, uh = (
            self.union_rect.x(),
            self.union_rect.y(),
            self.union_rect.width(),
            self.union_rect.height(),
        )
        hide_left   = ux - w - pad
        hide_right  = ux + uw + pad
        hide_top    = uy - h - pad
        hide_bottom = uy + uh + pad

        # Depending on dir_idx: 0=Left,1=Right,2=Top,3=Bottom
        if dir_idx == 0:
            return ((hide_left, t), (l, t))
        if dir_idx == 1:
            return ((hide_right, t), (l, t))
        if dir_idx == 2:
            return ((l, hide_top), (l, t))
        # dir_idx == 3:
        return ((l, hide_bottom), (l, t))
    
    def _slide_cfg(self, cfg, seg, into_view, cb=None):
        idx = DIRECTION_MAP.get(seg, 2)  # Use updated mapping; e.g., "Top-Middle" maps to top (2), "Bottom-Middle" to bottom (3)
        start, end = self._compute_positions(cfg["hwnd"], idx, cfg["orig"])
        if not into_view:
            start, end = end, start
        if cfg.get("animating", False):
            return
        cfg["animating"] = True
        def done():
            cfg["animating"] = False
            if into_view:
                win32gui.SetWindowPos(cfg["hwnd"], win32con.HWND_TOPMOST,
                                      end[0], end[1], 0, 0,
                                      win32con.SWP_NOSIZE | win32con.SWP_NOACTIVATE)
                try:
                    win32gui.SetForegroundWindow(cfg["hwnd"])
                except Exception:
                    pass
            if cb:
                cb()
        anim = Slider(
            cfg["hwnd"], start, end, cfg["steps"], cfg["interval"],
            use_ease=cfg["ease"], use_fade=cfg["fade"],
            into_view=into_view, activate_on_show=into_view, cb=done
        )
        self.animators.append(anim)
        anim.start()

    def _detect_segment(self, local_x, local_y, geom):
        w, h = geom.width(), geom.height()
        m = 3
        third = w / 3.0
        half = h / 2.0

        # Top edge (three segments)
        if local_y <= m:
            if   local_x < third:       return "Top-Left"
            elif local_x > 2 * third:   return "Top-Right"
            else:                        return "Top-Middle"

        # Bottom edge (three segments)
        if local_y >= h - m:
            if   local_x < third:       return "Bottom-Left"
            elif local_x > 2 * third:   return "Bottom-Right"
            else:                        return "Bottom-Middle"

        # Left edge (two segments)
        if local_x <= m:
            return "Left-Top" if local_y < half else "Left-Bottom"

        # Right edge (two segments)
        if local_x >= w - m:
            return "Right-Top" if local_y < half else "Right-Bottom"

        return None

    def _check_cursor_edge(self):
        if not self.enable_btn.isChecked():
            self.edge_dwell_start.clear()
            return

        pos = QCursor.pos()
        monitor = QGuiApplication.screenAt(pos)
        if not monitor:
            return

        geom = monitor.geometry()
        local_x = pos.x() - geom.x()
        local_y = pos.y() - geom.y()

        seg = self._detect_segment(local_x, local_y, geom)
        if not seg:
            # left the edge region
            self.edge_dwell_start.clear()
            return

        # Build the key (monitor index, segment)
        mon_idx = QGuiApplication.screens().index(monitor) + 1
        key = (mon_idx, seg)

        # Debounce/dwell logic
        now = time.time()
        threshold = self.edgeDelay.value() / 1000.0
        if key not in self.edge_dwell_start:
            self.edge_dwell_start[key] = now
        elif now - self.edge_dwell_start[key] >= threshold:
            if key not in self.triggered_segments:
                self.triggered_segments.add(key)
                self.segment_trigger_time[key] = now
                for cfg in self.window_cfg.get(key, []):
                    self._slide_cfg(cfg, seg, into_view=True)

        # Clean up any other pending dwells
        for other in list(self.edge_dwell_start):
            if other != key:
                del self.edge_dwell_start[other]
    
    def _check_focus(self):
        if not self.enable_btn.isChecked() or not self.triggered_segments:
            return
        fg = win32gui.GetForegroundWindow()
        keys_to_remove = []
        for key in self.triggered_segments:
            if time.time() - self.segment_trigger_time.get(key, 0) < 0.5:
                continue
            if not any(c["hwnd"] == fg for c in self.window_cfg.get(key, [])):
                for cfg in self.window_cfg.get(key, []):
                    self._slide_cfg(cfg, key[1], into_view=False)
                keys_to_remove.append(key)
        for key in keys_to_remove:
            self.triggered_segments.remove(key)
    
    def closeEvent(self, ev):
        if self.minimizeToTrayChk.isChecked():
            ev.ignore()
            self.hide()
            if self.trayIcon:
                self.trayIcon.showMessage(
                    "Minimized",
                    "Application is still running in the system tray.",
                    QSystemTrayIcon.MessageIcon.Information,
                    2000
                )
        else:
            if self.enable_btn.isChecked():
                self.enable_btn.setChecked(False)
            for key in list(self.window_cfg.keys()):
                for cfg in self.window_cfg[key]:
                    l, t, r, b = cfg["orig"]
                    win32gui.SetWindowPos(cfg["hwnd"], None, l, t, 0, 0,
                                          win32con.SWP_NOSIZE | win32con.SWP_NOZORDER)
                    self._show_taskbar(cfg["hwnd"])
            ev.accept()
